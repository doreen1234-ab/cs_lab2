#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
#include <sys/types.h>

int main() {
    int i;
    pid_t pid;
    
    printf("Parent Process Started. PID: %d\n", getpid());
    printf("Creating 5 child processes...\n\n");
    
    for (i = 1; i <= 5; i++) {
        pid = fork();  
        
        if (pid < 0) {
           
            printf("Error: Fork failed\n");
            exit(1);
        }
        else if (pid == 0) {
            
            printf("Child %d created with PID: %d\n", i, getpid());
            
            
            sleep(i * 1);  // Child 1 sleeps 1 sec, Child 2 sleeps 2 sec, etc.
            
            printf("Child %d (PID: %d) is exiting\n", i, getpid());
            exit(0);  // Child exits
        }

    }
    

    printf("\nParent waiting for all children to finish...\n");
    
    for (i = 1; i <= 5; i++) {
        int status;
        pid_t child_pid = wait(&status);  // Wait for any child to finish
        
        if (child_pid > 0) {
            printf("Parent: Cleaned up child with PID: %d\n", child_pid);
            
            if (WIFEXITED(status)) {
                printf("   Child exited normally with status: %d\n", WEXITSTATUS(status));
            }
        }
    }
    
    printf("\nAll children cleaned up. Parent exiting.\n");
    return 0;
}
